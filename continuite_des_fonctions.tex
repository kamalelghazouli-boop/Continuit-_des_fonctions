\section{Continuité d'une fonction}
% =====================================================================

\subsection{Définition}

Soit $f$ une fonction définie sur un intervalle $I$ de $\R$.

Intuitivement, dire que $f$ est continue sur $I$ signifie que sa courbe
représentative peut être tracée \emph{en un seul morceau}, sans lever le crayon :
la courbe ne présente aucun saut, aucun trou.

Mathématiquement, cela se traduit de la manière suivante.

\begin{definition}{Continuité en un point, sur un intervalle}{continuite}
Soit $f$ une fonction définie sur un intervalle $I$ de $\R$ et $a$ un réel de $I$
(qui n'est pas une borne de $I$).
\begin{itemize}
  \item On dit que $f$ est \textbf{continue en $a$} lorsque la limite à gauche
  en $a$ et la limite à droite en $a$ sont \textbf{égales} à $f(a)$ :
  \[
    \limg{a}{f(x)} \;=\; \limd{a}{f(x)} \;=\; f(a).
  \]
  Sinon, on dit que $f$ est \textbf{discontinue} en $a$.
  \item On dit que $f$ est \textbf{continue sur $I$} lorsque $f$ est continue
  en \textbf{tout réel $a$ de $I$}.
\end{itemize}
\end{definition}

\begin{remarque}[Pour vérifier la continuité en $a$]
On calcule séparément les trois nombres suivants, puis on les compare :
\begin{enumerate}[label=\textcolor{bleu}{\textbf{\arabic*.}}]
  \item l'image $f(a)$ ;
  \item la limite à gauche $\limg{a}{f(x)}$ ;
  \item la limite à droite $\limd{a}{f(x)}$.
\end{enumerate}
Si ces trois nombres sont égaux, $f$ est continue en $a$. Dans ce cas, on écrit
simplement $\limite{x\to a}{f(x)}=f(a)$.
\end{remarque}

\begin{remarque}[Cas d'une borne]
Si $a$ est une borne de $I$ (par exemple $I=\intff{a}{b}$), on ne considère que la
limite à droite en $a$ : $f$ est continue en $a$ si $\limd{a}{f(x)}=f(a)$.

De même en $b$, avec la limite à gauche.
\end{remarque}

\subsection{Exemples d'application}

\begin{exemple}{Continuité en un point d'une fonction polynôme}{}
Soit $f$ la fonction définie sur $\R$ par $f(x)=x^2-3x+1$.
Montrons que $f$ est continue en $2$.

\begin{etapes}
  \item Image : $f(2)=2^2-3\times 2+1=-1$.
  \item Limite à gauche : par somme et produit de limites,
        $\limg{2}{f(x)}=2^2-3\times 2+1=-1$.
  \item Limite à droite : de même, $\limd{2}{f(x)}=-1$.
\end{etapes}

\textbf{Conclusion :} $\limg{2}{f(x)}=\limd{2}{f(x)}=f(2)=-1$, donc la fonction
$f$ est continue en $2$.
\end{exemple}

\begin{exemple}{Fonction définie par morceaux -- recherche d'un paramètre}{}
Soit $m$ un réel et $f$ la fonction définie sur $\R$ par
\[
  f(x)=\begin{cases}
    2x+1 & \text{si } x<1,\\[2pt]
    x^2+m & \text{si } x\geqslant 1.
  \end{cases}
\]
Sur $\intoo{-\infty}{1}$ et sur $\intoo{1}{+\infty}$, $f$ coïncide avec une
fonction polynôme, donc elle y est continue (voir partie~\ref{sec:usuelles}).

Étudions la continuité en $1$.
\begin{etapes}
  \item Image : $f(1)=1^2+m=1+m$.
  \item Limite à gauche : $\limg{1}{f(x)}=\limg{1}{(2x+1)}=3$.
  \item Limite à droite : $\limd{1}{f(x)}=\limd{1}{(x^2+m)}=1+m$.
  \item Comparaison : $f$ est continue en $1$ si et seulement si $3=1+m$,
        c'est-à-dire $m=2$.
\end{etapes}

\textbf{Conclusion :} $f$ est continue sur $\R$ si et seulement si $m=2$.
\end{exemple}

\begin{exemple}{Une fonction discontinue : la fonction partie entière}{}
Pour tout réel $x$, il existe un unique entier relatif $n$ tel que
$n\leqslant x<n+1$ ; on l'appelle \textbf{partie entière} de $x$ et on le note
$E(x)$ (en Python : \texttt{floor(x)}).

Par exemple : $E(2{,}7)=2$, \ $E(3)=3$, \ $E(-1{,}2)=-2$.

\begin{minipage}[c]{0.55\linewidth}
Étudions la continuité en $1$.
\begin{etapes}
  \item Image : $E(1)=1$.
  \item Pour $x\in\intfo{0}{1}$, $E(x)=0$, donc $\limg{1}{E(x)}=0$.
  \item Pour $x\in\intfo{1}{2}$, $E(x)=1$, donc $\limd{1}{E(x)}=1$.
\end{etapes}

\textbf{Conclusion :} $\limg{1}{E(x)}\neq\limd{1}{E(x)}$, donc la fonction $E$
\textbf{n'est pas continue en $1$}.
\end{minipage}\hfill

\section{Continuité des fonctions usuelles et opérations}\label{sec:usuelles}

\subsection{Continuité des fonctions de référence}

On a le tableau de continuité des fonctions de référence suivant (résultats admis) :

\begin{center}
\setstretch{1}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|>{\centering\arraybackslash}m{6.6cm}|>{\centering\arraybackslash}m{6.6cm}|}
\hline
\rowcolor{bleuclair}\textbf{Fonctions usuelles} & \textbf{Intervalle(s) de continuité} \\ \hline
$f(x)=x^n$, \quad $n\in\N$ & $\R$ \\ \hline
$f(x)=\dfrac{1}{x^n}$, \quad $n\in\N^*$ & $\intoo{-\infty}{0}$ et $\intoo{0}{+\infty}$ \\ \hline
$f(x)=\sqrt{x}$ & $\intfo{0}{+\infty}$ \\ \hline
$f(x)=\lvert x\rvert$ & $\R$ \\ \hline
$f(x)=\e^x$ & $\R$ \\ \hline
$f(x)=\sin x$ \ et \ $f(x)=\cos x$ & $\R$ \\ \hline

\end{tabular}
\end{center}

\begin{remarque}
Il en découle (avec le théorème suivant) que toute \textbf{fonction polynôme}
est continue sur $\R$, et que toute \textbf{fonction rationnelle} (quotient de
deux polynômes) est continue sur chacun des intervalles où elle est définie.
\end{remarque}

\subsection{Opérations sur les fonctions continues}

\begin{theoreme}{Opérations}{operations}
Soit $k$ un réel. Soit $f$ et $g$ deux fonctions définies sur un intervalle $I$
et soit $a\in I$. Si $f$ et $g$ sont continues en $a$, alors :
\begin{itemize}
  \item $f+g$, \ $k\times f$ \ et \ $f\times g$ sont continues en $a$ ;
  \item $\dfrac{f}{g}$ est continue en $a$, à condition que $g(a)\neq 0$.
\end{itemize}
\textbf{Composition :} si $f$ est continue en $a$ et si $g$ est définie sur un
intervalle $J$ contenant $f(I)$ et continue en $f(a)$, alors $g\circ f$ est
continue en $a$.

Ces résultats s'étendent aux fonctions continues sur un intervalle.
\end{theoreme}

\subsection{Exemples d'application}

\begin{exemple}{Fonction rationnelle}{}
Soit $f$ la fonction définie par $f(x)=\dfrac{3x^2-1}{x-2}$.
\begin{etapes}
  \item Le numérateur et le dénominateur sont des fonctions polynômes,
        donc continues sur $\R$.
  \item Le dénominateur s'annule uniquement en $x=2$.
\end{etapes}
\textbf{Conclusion :} par quotient, $f$ est continue sur $\intoo{-\infty}{2}$
et sur $\intoo{2}{+\infty}$.
\end{exemple}

\begin{exemple}{Composition}{}
Soit $g$ la fonction définie sur $\R$ par $g(x)=\sqrt{x^2+1}$.
\begin{etapes}
  \item La fonction $u:x\longmapsto x^2+1$ est continue sur $\R$ (polynôme).
  \item Pour tout réel $x$, $u(x)\geqslant 1>0$, donc $u(\R)\subset\intfo{0}{+\infty}$.
  \item La fonction racine carrée est continue sur $\intfo{0}{+\infty}$.
\end{etapes}
\textbf{Conclusion :} par composition, $g=\sqrt{\phantom{x}}\circ u$ est continue
sur $\R$.
\end{exemple}

\begin{exemple}{Produit et composition}{}
Soit $h$ la fonction définie sur $\R$ par $h(x)=\e^{-x}\cos(2x)$.
\begin{etapes}
  \item $x\mapsto -x$ et $\exp$ sont continues sur $\R$, donc, par composition,
        $x\mapsto \e^{-x}$ est continue sur $\R$.
  \item $x\mapsto 2x$ et $\cos$ sont continues sur $\R$, donc, par composition,
        $x\mapsto\cos(2x)$ est continue sur $\R$.
\end{etapes}
\textbf{Conclusion :} par produit, $h$ est continue sur $\R$.
\end{exemple}

\section{Continuité et dérivabilité}
% =====================================================================

\subsection{Théorème}

\begin{theoreme}{Dérivabilité implique continuité}{derivcont}
Soit $f$ une fonction définie sur un intervalle $I$ et soit $a\in I$.
\begin{itemize}
  \item Si $f$ est \textbf{dérivable en $a$}, alors la fonction $f$ est
        \textbf{continue en $a$}.
  \item Si $f$ est \textbf{dérivable sur $I$}, alors la fonction $f$ est
        \textbf{continue sur $I$}.
\end{itemize}
\end{theoreme}

\begin{demo}
Supposons $f$ dérivable en $a$.

Pour tout réel $h\neq 0$ tel que $a+h\in I$, on peut écrire :
\[
  f(a+h) = f(a) + h\times\frac{f(a+h)-f(a)}{h}.
\]
Or $\limite{h\to 0}{\dfrac{f(a+h)-f(a)}{h}}=f'(a)$, qui est un nombre réel, et
$\limite{h\to 0}{h}=0$.

Donc, par produit : $\limite{h\to 0}{\left(h\times\dfrac{f(a+h)-f(a)}{h}\right)}
=0\times f'(a)=0$.

Ainsi $\limite{h\to 0}{f(a+h)}=f(a)$, que $h$ tende vers $0$ par valeurs
négatives ou positives : les limites à gauche et à droite de $f$ en $a$ sont
égales à $f(a)$.

Donc $f$ est continue en $a$.
\end{demo}

\subsection{La réciproque est fausse}

\begin{minipage}[c]{0.6\linewidth}
La réciproque de ce théorème est fausse : une fonction peut être continue en un
point $a$ sans pour autant y être dérivable.

C'est le cas de la fonction valeur absolue : $f(x)=\lvert x\rvert$.
\end{minipage}\hfill
\begin{minipage}[c]{0.36\linewidth}
\centering

  \draw[gris!30] (-3.5,-0.8) grid (3.5,3.8);

  \foreach \k in {1,2,3}{\node[left, font=\scriptsize] at (0,\k) {$\k$};}
  \node[below left, font=\scriptsize] at (0,0) {$O$};
  \draw[thick, rouge] (-3.6,3.6) -- (0,0) -- (3.6,3.6);
  \draw[-{Stealth}, very thick, bleu] (0,0) -- (-1,1);

{\small\setstretch{1} La courbe est en « un seul morceau » \\ et possède un point
anguleux en $O$.\par}
\end{minipage}

\begin{demo}
\textbf{$f$ est continue en $0$.}

Pour $x<0$, $f(x)=-x$, donc $\limg{0}{f(x)}=0$.

Pour $x>0$, $f(x)=x$, donc $\limd{0}{f(x)}=0$.

Comme $f(0)=0$, on a $\limg{0}{f(x)}=\limd{0}{f(x)}=f(0)$ : $f$ est continue en $0$.

\textbf{$f$ n'est pas dérivable en $0$.}

Pour $h\neq 0$ :
$\dfrac{f(0+h)-f(0)}{h}=\dfrac{\lvert h\rvert}{h}=
  \begin{cases} 1 & \text{si } h>0,\\ -1 & \text{si } h<0.\end{cases}$

Le taux de variation admet deux limites différentes à droite ($1$) et à gauche
($-1$) en $0$ : il n'a donc pas de limite en $0$, et $f$ n'est pas dérivable en $0$.
\end{demo}

\subsection{Exemples d'application}

\begin{exemple}{Utiliser la dérivabilité pour justifier la continuité}{}
Soit $f$ la fonction définie sur $\R$ par $f(x)=(2x-1)\e^{x}$.
\begin{etapes}
  \item $x\mapsto 2x-1$ et $x\mapsto\e^x$ sont dérivables sur $\R$, donc, par
        produit, $f$ est dérivable sur $\R$.
  \item D'après le théorème~\ref{thm:derivcont}, toute fonction dérivable sur $\R$
        est continue sur $\R$.
\end{etapes}
\textbf{Conclusion :} $f$ est continue sur $\R$.
\end{exemple}

\begin{exemple}{La fonction racine carrée en $0$}{}
Soit $f$ la fonction définie sur $\intfo{0}{+\infty}$ par $f(x)=\sqrt{x}$.
\begin{etapes}
  \item Continuité en $0$ ($0$ est une borne, on regarde la limite à droite) :
        $\limd{0}{\sqrt{x}}=0=f(0)$, donc $f$ est continue en $0$.
  \item Dérivabilité en $0$ : pour $h>0$,
  \[
    \frac{f(0+h)-f(0)}{h}=\frac{\sqrt{h}}{h}=\frac{1}{\sqrt{h}}
    \xrightarrow[h\to 0^+]{}+\infty .
  \]
  Le taux de variation n'a pas de limite finie : $f$ n'est pas dérivable en $0$.
\end{etapes}
\textbf{Interprétation graphique :} la courbe de la fonction racine carrée admet en
$O$ une tangente verticale.
\end{exemple}

\subsection{Théorème des valeurs intermédiaires}

\begin{theoreme}{Théorème des valeurs intermédiaires (TVI) -- cas général}{tvi}
Soit $f$ une fonction \textbf{continue} sur un intervalle $\intff{a}{b}$ et $k$
un réel compris entre $f(a)$ et $f(b)$.

Alors il existe (au moins) un réel $c$ dans $\intff{a}{b}$ tel que $f(c)=k$.

Autrement dit, l'équation $f(x)=k$ admet au moins une solution dans $\intff{a}{b}$.
\end{theoreme}

\begin{remarque}
Ce théorème (admis) résulte du fait que \textbf{l'image d'un intervalle} de $\R$
par une fonction continue est \textbf{un intervalle} de $\R$.
\end{remarque}

Ici, il existe 3 valeurs : $c_1$, $c_2$ et $c_3$.

\begin{exemple}{Existence d'une solution}{}
Montrons que l'équation $\cos x = x$ admet au moins une solution dans
$\intff{0}{\frac{\pi}{2}}$.

Posons $h(x)=\cos x - x$.
\begin{etapes}
  \item Continuité : $h$ est continue sur $\intff{0}{\frac{\pi}{2}}$
        (différence de fonctions continues).
  \item Valeurs aux bornes : $h(0)=1>0$ \ et \
        $h\!\left(\frac{\pi}{2}\right)=0-\frac{\pi}{2}<0$.
  \item Donc $0$ est compris entre $h\big(\frac{\pi}{2}\big)$ et $h(0)$.
\end{etapes}
\textbf{Conclusion :} d'après le TVI, il existe au moins un réel
$c\in\intff{0}{\frac{\pi}{2}}$ tel que $h(c)=0$, c'est-à-dire $\cos c=c$.
\end{exemple}

\subsection{Extension aux intervalles ouverts}

Il est également possible d'utiliser les limites dans le théorème des valeurs
intermédiaires.

\begin{theoreme}{TVI avec des limites}{tvilim}
Soit $f$ une fonction continue sur un intervalle $\intoo{a}{b}$ (avec $a$ et $b$
réels ou infinis) telle que les limites $\limd{a}{f(x)}$ et $\limg{b}{f(x)}$
existent (finies ou infinies).

Soit $k$ un réel strictement compris entre ces deux limites.

Alors il existe (au moins) un réel $c$ dans $\intoo{a}{b}$ tel que $f(c)=k$.
\end{theoreme}

\begin{exemple}{}{}
Soit $f$ la fonction polynôme de degré $3$ définie sur $\R$ par $f(x)=x^3+3$.
\begin{etapes}
  \item Continuité : $f$ est continue sur $\R$ (fonction polynôme).
  \item Limites : $\limite{x\to+\infty}{f(x)}=+\infty$ \ et \
        $\limite{x\to-\infty}{f(x)}=-\infty$.
  \item Or $0\in\intoo{-\infty}{+\infty}$.
\end{etapes}
\textbf{Conclusion :} d'après le théorème des valeurs intermédiaires, il existe
un réel $x$ tel que $f(x)=0$.
\end{exemple}

\subsection{Cas des fonctions strictement monotones}

\begin{corollaire}{Corollaire du TVI -- théorème de la bijection}{bijection}
Soit $f$ une fonction \textbf{continue} et \textbf{strictement monotone} sur
$I=\intff{a}{b}$.

Pour tout réel $k$ compris entre $f(a)$ et $f(b)$, l'équation $f(x)=k$ admet
\textbf{une unique solution} dans $I$.
\end{corollaire}

\begin{minipage}[c]{0.55\linewidth}
\begin{demo}[unbreakable]
\textbf{Existence :} c'est le TVI.

\textbf{Unicité :} supposons qu'il existe deux solutions $c<c'$.

Par stricte monotonie, $f(c)<f(c')$ ou $f(c)>f(c')$.

Cela contredit $f(c)=f(c')=k$.
\end{demo}

\begin{remarque}
\begin{itemize}
  \item Ce théorème est aussi appelé \textbf{théorème de la bijection}.
  \item On généralise ce théorème à l'intervalle ouvert $I=\intoo{a}{b}$.
        Le réel $k$ doit alors être strictement compris entre
        $\limd{a}{f(x)}$ et $\limg{b}{f(x)}$.
  \item Lorsque $k=0$, on pourra montrer que $f(a)\times f(b)<0$.
  \item \textbf{Convention :} dans un tableau de variations, une flèche oblique
        traduit la \textbf{continuité} et la \textbf{stricte monotonie} de la
        fonction sur l'intervalle correspondant.
\end{itemize}
\end{remarque}

\begin{methode}[Étudier une équation $f(x)=k$ : existence, unicité, encadrement]
\begin{enumerate}
  \item Justifier que $f$ est \textbf{continue} sur l'intervalle $I$ considéré.
  \item Justifier que $f$ est \textbf{strictement monotone} sur $I$
        (souvent par l'étude du signe de $f'$).
  \item Calculer les images (ou les limites) aux bornes de $I$ et vérifier que
        $k$ est compris entre ces valeurs.
  \item Conclure avec le corollaire du TVI : l'équation $f(x)=k$ admet une
        unique solution $\alpha$ dans $I$.
  \item \textbf{Encadrer} $\alpha$ à l'aide de la calculatrice (balayage,
        tableau de valeurs) ou d'un algorithme (dichotomie).
\end{enumerate}
\end{methode}

\subsection{Exemples d'application}

\begin{exemple}{Existence, unicité et encadrement d'une solution}{}
Soit $f$ la fonction définie sur $\R$ par $f(x)=x^3+x-1$.

Montrons que l'équation $f(x)=0$ admet une unique solution $\alpha$ dans $\R$,
puis donnons-en un encadrement d'amplitude $10^{-2}$.

\begin{etapes}
  \item Continuité : $f$ est une fonction polynôme, donc continue et dérivable
        sur $\R$.
  \item Stricte monotonie : pour tout réel $x$, $f'(x)=3x^2+1>0$, donc $f$ est
        strictement croissante sur $\R$.
  \item Limites aux bornes :
        $\limite{x\to-\infty}{f(x)}=\limite{x\to-\infty}{x^3}=-\infty$ \ et \
        $\limite{x\to+\infty}{f(x)}=\limite{x\to+\infty}{x^3}=+\infty$.

\begin{exemple}{Dénombrer les solutions à l'aide du tableau de variations}{}
Soit $g$ la fonction définie sur $\R$ par $g(x)=x^3-3x+1$.

Déterminons le nombre de solutions de l'équation $g(x)=0$.

\begin{etapes}
  \item Dérivée : $g$ est continue et dérivable sur $\R$, et
        $g'(x)=3x^2-3=3(x-1)(x+1)$.
  \item Valeurs et limites : $g(-1)=3$, \ $g(1)=-1$, \
        $\limite{x\to-\infty}{g(x)}=-\infty$ \ et \
        $\limite{x\to+\infty}{g(x)}=+\infty$.

\subsection{Algorithme de dichotomie}

On suppose que $f$ est continue et strictement monotone sur $\intff{a}{b}$, avec
$f(a)\times f(b)<0$.

L'équation $f(x)=0$ admet donc une unique solution $\alpha\in\intff{a}{b}$.

Le principe de la \textbf{dichotomie} consiste à couper l'intervalle en deux à
chaque étape :
\begin{etapes}
  \item On calcule le milieu $m=\dfrac{a+b}{2}$.
  \item Si $f(a)\times f(m)\leqslant 0$, alors $\alpha\in\intff{a}{m}$ et on
        remplace $b$ par $m$ ; sinon $\alpha\in\intff{m}{b}$ et on remplace $a$
        par $m$.
  \item On recommence tant que l'amplitude $b-a$ est supérieure à la précision
        $p$ souhaitée.
\end{etapes}
Après $n$ étapes, l'amplitude de l'intervalle est $\dfrac{b-a}{2^n}$.

\begin{minipage}[c]{0.52\linewidth}
\begin{lstlisting}
def dichotomie(f, a, b, p):
    while b - a > p:
        m = (a + b) / 2
        if f(a) * f(m) <= 0:
            b = m
        else:
            a = m
    return a, b

def f(x):
    return x**3 + x - 1

print(dichotomie(f, 0, 1, 0.01))
\end{lstlisting}
\end{minipage}\hfill

\begin{exemple}[unbreakable]{}{}
Avec $f(x)=x^3+x-1$ sur $\intff{0}{1}$ et $p=0{,}01$, le programme affiche :
\begin{center}
\texttt{(0.6796875, 0.6875)}
\end{center}
Ainsi $0{,}679<\alpha<0{,}688$ après $7$ étapes, car
$\frac{1}{2^7}\approx 0{,}0078<0{,}01$.

Ce résultat est cohérent avec l'encadrement obtenu par balayage.
\end{exemple}

Fonctions usuelles & Polynômes, $\e^x$, $\sin$, $\cos$, $\lvert x\rvert$ continues sur $\R$ ; $\sqrt{x}$ sur $\intfo{0}{+\infty}$ ; rationnelles sur leurs intervalles de définition. \\ \hline
